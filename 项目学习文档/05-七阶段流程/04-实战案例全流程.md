## 📋 案例背景

### 项目信息

**项目名称：** 用户每日签到功能

**业务目标：**
- 提升日活跃用户数（DAU）
- 增加用户粘性
- 培养用户每日登录习惯

**项目规模：**
- 开发周期：1 周
- 团队成员：1 名后端 + 1 名前端
- 技术栈：Node.js + Express + MySQL + React

```typescript
import { injectable } from 'typedi';
import { UserSignInRepository } from '../repositories/UserSignInRepository';
import { SignInEvent } from '../events/SignInEvent';

@injectable()
export class SignInService {
  constructor(
    private userSignInRepo: UserSignInRepository,
    private eventEmitter: EventEmitter
  ) {}

  async signIn(userId: number, ipAddress: string): Promise<SignInResult> {
    // 检查今日是否已签到
    const hasSignedIn = await this.hasSignedInToday(userId);
    if (hasSignedIn) {
      throw new DuplicateSignInError('今日已签到');
    }

    // 计算连续天数
    const consecutiveDays = await this.calculateConsecutiveDays(userId);

    // 计算获得积分
    const basePoints = 10;
    const bonusPoints = this.calculateBonus(consecutiveDays);
    const totalPoints = basePoints + bonusPoints;

    // 保存签到记录
    const record = await this.userSignInRepo.create({
      userId,
      signInDate: new Date(),
      consecutiveDays,
      pointsEarned: totalPoints,
      ipAddress
    });

    // 发布事件（触发积分发放）
    this.eventEmitter.emit('signin.completed', {
      userId,
      points: totalPoints,
      consecutiveDays,
      recordId: record.id
    });

    return {
      success: true,
      points: totalPoints,
      consecutiveDays,
      message: `签到成功！获得${totalPoints}积分，已连续签到${consecutiveDays}天`
    };
  }

  private calculateBonus(days: number): number {
    if (days >= 30) return 300;
    if (days >= 7) return 50;
    return 0;
  }
}
```
